# Editable Profiles Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Let teachers/admins edit name, contact info, and photo on the web, and students edit contact info and photo on mobile, through one shared `/api/v1/me/` profile API.

**Architecture:** Extend the existing Django `MeView` with PATCH and add a `MeAvatarView` for multipart photo upload/delete. The React portal gets a `/profile` page and header/drawer links; the Flutter app gets an Edit profile screen backed by `AppState`. Avatar files live under the existing `media/` serving setup.

**Tech Stack:** Django 5.2 + DRF 3.16 + SimpleJWT, Pillow (new), React 19 + Vite 7 + Tailwind 4, Flutter 3.44/Dart 3.12 + Provider + `package:http` (already a dependency) + `image_picker` (new).

**Spec:** `docs/superpowers/specs/2026-11-20-editable-profiles-design.md`

## Global Constraints

- Usernames/LRNs, roles, and passwords are never touched by profile endpoints; student display names stay school-managed (server-enforced 400 on student `first_name`/`last_name`).
- Avatars: `.png`/`.jpg`/`.jpeg`/`.webp` only (no SVG), max 2 MiB (`2 * 1024 * 1024`), verified by Pillow decode; randomized UUID filenames under `media/avatars/`; old file deleted on replace/remove.
- Audit entries via `record_audit` with old/new values only for changed fields; never log credentials or image bytes.
- UI uses the approved powder-blue system: white cards, `--color-primary-container: #263c52` actions, 44px minimum targets, visible focus states. Do not change grading workflows.
- Commit only the files listed in each task. Never stage unrelated working-tree changes (`mobile/android/*`, deploy scripts, `web/.ui-audit/`, other UI-refresh edits already in the tree).
- Backend commands run from `backend/` with `./.venv/Scripts/python.exe`; web from `web/`; mobile from `mobile/`.
- Mobile base URL is `String.fromEnvironment('API_BASE_URL', defaultValue: 'http://10.0.2.2:8000')` in `mobile/lib/main.dart:13`; the deployed build script `deploy/apk-build.cmd` injects `https://grades.darkzhub.asia`.

class ProfileUpdateSerializer(serializers.ModelSerializer):
    class Meta:
        model = User
        fields = ["first_name", "last_name", "email", "phone"]

    def validate(self, attrs):
        if self.instance.is_student and ("first_name" in attrs or "last_name" in attrs):
            raise serializers.ValidationError(
                "Student names are managed by the school. Contact your adviser for corrections."
            )
        return attrs

    def update(self, instance, validated_data):
        changed = {
            field: value
            for field, value in validated_data.items()
            if getattr(instance, field) != value
        }
        if changed:
            old = {field: getattr(instance, field) for field in changed}
            instance = super().update(instance, validated_data)
            record_audit(
                actor=instance,
                action="profile.update",
                entity="user",
                entity_id=instance.id,
                old_values=old,
                new_values=changed,
            )
        return instance
```

class AvatarUploadSerializer(serializers.Serializer):
    avatar = serializers.FileField()

    def validate_avatar(self, value):
        allowed = (".png", ".jpg", ".jpeg", ".webp")
        if not value.name.lower().endswith(allowed):
            raise serializers.ValidationError("Photo must be a PNG, JPG or WEBP file.")
        if value.size > 2 * 1024 * 1024:
            raise serializers.ValidationError("Photo must be smaller than 2 MB.")
        try:
            from PIL import Image

            Image.open(value).verify()
        except Exception:
            raise serializers.ValidationError("The uploaded file is not a valid image.")
        finally:
            value.seek(0)
        return value
```

class MeAvatarView(APIView):
    permission_classes = [IsAuthenticatedActive]
    parser_classes = [MultiPartParser, FormParser]

    def post(self, request):
        serializer = AvatarUploadSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        user = request.user
        old_path = user.avatar.name if user.avatar else None
        if user.avatar:
            user.avatar.delete(save=False)
        user.avatar = serializer.validated_data["avatar"]
        user.save(update_fields=["avatar"])
        record_audit(
            actor=user,
            action="profile.avatar",
            entity="user",
            entity_id=user.id,
            old_values={"avatar": old_path},
            new_values={"avatar": user.avatar.name},
        )
        return Response({"avatar_url": absolute_file_url(request, user.avatar)})

    def delete(self, request):
        user = request.user
        if user.avatar:
            old_path = user.avatar.name
            user.avatar.delete(save=False)
            user.avatar = None
            user.save(update_fields=["avatar"])
            record_audit(
                actor=user,
                action="profile.avatar",
                entity="user",
                entity_id=user.id,
                old_values={"avatar": old_path},
                new_values={"avatar": None},
            )
        return Response({"avatar_url": None})
```

export async function apiForm(path, { method = "POST", body } = {}) {
  const doFetch = () => {
    const headers = {};
    if (storage.access) headers.Authorization = `Bearer ${storage.access}`;
    return fetch(`${BASE}${path}`, { method, headers, body });
  };

  let res = await doFetch();
  if (res.status === 401) {
    if (await tryRefresh()) {
      res = await doFetch();
    } else {
      storage.clear();
      onSessionExpired();
      throw new ApiError(401, { error: { code: "not_authenticated", message: "Session expired. Please log in again." } });
    }
  }
  const data = await res.json().catch(() => null);
  if (!res.ok) throw new ApiError(res.status, data);
  return data;
}

export const profileApi = {
  update(details) {
    return api("/me/", { method: "PATCH", body: details });
  },
  uploadAvatar(file) {
    const body = new FormData();
    body.append("avatar", file, file.name);
    return apiForm("/me/avatar/", { method: "POST", body });
  },
  removeAvatar() {
    return api("/me/avatar/", { method: "DELETE" });
  },
};
```

import { useEffect, useRef, useState } from "react";
import { Link } from "react-router-dom";
import { profileApi } from "../api/client";
import { useAuth } from "../auth/AuthContext";
import { Btn, Card, ErrorBanner, Field, PageHeader, SectionCard, Spinner, inputClass, useToast } from "../components/ui";

const ACCEPTED_TYPES = ["image/png", "image/jpeg", "image/webp"];

export default function Profile() {
  const { user, loading, refreshUser } = useAuth();
  const toast = useToast();
  const fileInput = useRef(null);
  const [form, setForm] = useState({ first_name: "", last_name: "", email: "", phone: "" });
  const [photoFile, setPhotoFile] = useState(null);
  const [preview, setPreview] = useState(null);
  const [error, setError] = useState(null);
  const [saving, setSaving] = useState(false);
  const [uploading, setUploading] = useState(false);

  useEffect(() => {
    if (user) {
      setForm({
        first_name: user.first_name || "",
        last_name: user.last_name || "",
        email: user.email || "",
        phone: user.phone || "",
      });
    }
  }, [user]);

  useEffect(() => {
    if (!photoFile) {
      setPreview(null);
      return;
    }
    const url = URL.createObjectURL(photoFile);
    setPreview(url);
    return () => URL.revokeObjectURL(url);
  }, [photoFile]);

  if (loading || !user) return <Spinner />;

  const initials = (user.display_name || user.username || "?")
    .split(/[\s,]+/)
    .filter(Boolean)
    .slice(0, 2)
    .map((part) => part[0].toUpperCase())
    .join("");
  const shownAvatar = preview || user.avatar_url;

  function pickPhoto(event) {
    const file = event.target.files?.[0];
    event.target.value = "";
    setError(null);
    if (!file) return;
    if (!ACCEPTED_TYPES.includes(file.type)) {
      setError({ message: "Photo must be a PNG, JPG or WEBP file." });
      return;
    }
    if (file.size > 2 * 1024 * 1024) {
      setError({ message: "Photo must be smaller than 2 MB." });
      return;
    }
    setPhotoFile(file);
  }

  async function uploadPhoto() {
    if (!photoFile) return;
    setUploading(true);
    setError(null);
    try {
      await profileApi.uploadAvatar(photoFile);
      setPhotoFile(null);
      await refreshUser();
      toast.success("Profile photo updated.");
    } catch (e) {
      setError(e);
    } finally {
      setUploading(false);
    }
  }

  async function removePhoto() {
    setUploading(true);
    setError(null);
    try {
      await profileApi.removeAvatar();
      setPhotoFile(null);
      await refreshUser();
      toast.success("Profile photo removed.");
    } catch (e) {
      setError(e);
    } finally {
      setUploading(false);
    }
  }

  async function saveDetails(event) {
    event.preventDefault();
    setSaving(true);
    setError(null);
    try {
      await profileApi.update(form);
      await refreshUser();
      toast.success("Profile details saved.");
    } catch (e) {
      setError(e);
    } finally {
      setSaving(false);
    }
  }

  return (
    <>
      <PageHeader
        eyebrow="Account"
        title="My Profile"
        sub="Your personal details and profile photo. Usernames and roles are managed by the administrator."
      />
      {error && <div className="mb-space-sm"><ErrorBanner error={error} /></div>}

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-space-lg items-start">
        <Card className="p-space-lg flex flex-col items-center text-center gap-space-md">
          {shownAvatar ? (
            <img src={shownAvatar} alt="Profile photo" className="w-24 h-24 rounded-full object-cover border border-outline-variant" />
          ) : (
            <div className="w-24 h-24 rounded-full bg-primary-container text-on-primary flex items-center justify-center text-2xl font-bold">
              {initials}
            </div>
          )}
          <div>
            <p className="font-headline-sm text-headline-sm text-on-surface">{user.display_name}</p>
            <p className="font-body-sm text-body-sm text-on-surface-variant">
              {user.username} · <span className="capitalize">{user.role}</span>
            </p>
          </div>
          <input ref={fileInput} type="file" accept="image/png,image/jpeg,image/webp" className="hidden" onChange={pickPhoto} aria-label="Choose profile photo" />
          <div className="flex flex-col w-full gap-space-sm">
            <Btn variant="secondary" icon="photo_camera" onClick={() => fileInput.current?.click()} disabled={uploading}>
              {photoFile ? "Choose a different photo" : "Choose photo"}
            </Btn>
            {photoFile && (
              <Btn icon="upload" onClick={uploadPhoto} disabled={uploading}>
                {uploading ? "Uploading…" : "Upload photo"}
              </Btn>
            )}
            {user.avatar_url && !photoFile && (
              <Btn variant="secondary" icon="delete" onClick={removePhoto} disabled={uploading}>
                Remove photo
              </Btn>
            )}
          </div>
          <p className="font-body-sm text-body-sm text-on-surface-variant">PNG, JPG or WEBP, up to 2 MB.</p>
        </Card>

        <SectionCard title="Personal details" sub="Shown in the portal header and on reports." className="lg:col-span-2">

    final access = await storage.getAccess();
    final request = await _open(method, path);
    try {

    } on ApiException catch (e) {

  Future<http.Response> _uploadAvatarOnce(File file, String access) async {
    final request = http.MultipartRequest('POST', Uri.parse('$baseUrl/api/v1/me/avatar/'));
    request.headers['Authorization'] = 'Bearer $access';
    request.files.add(await http.MultipartFile.fromPath('avatar', file.path));
    final streamed = await request.send().timeout(const Duration(seconds: 30));
    return http.Response.fromStream(streamed);
  }

  Future<String?> uploadAvatar(File file) async {
    var access = await storage.getAccess() ?? '';
    var response = await _uploadAvatarOnce(file, access);
    if (response.statusCode == 401 && await tryRefresh()) {
      access = await storage.getAccess() ?? '';
      response = await _uploadAvatarOnce(file, access);
    }
    Map<String, dynamic> data = {};
    try {
      data = jsonDecode(response.body) as Map<String, dynamic>;
    } catch (_) {}
    if (response.statusCode < 200 || response.statusCode >= 300) {
      final error = data['error'];
      throw ApiException(
        response.statusCode,
        error is Map ? (error['message'] as String? ?? 'Upload failed.') : 'Upload failed.',
      );
    }
    return data['avatar_url'] as String?;
  }
